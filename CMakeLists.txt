cmake_minimum_required(VERSION 3.16)
project(LINK4BRICK VERSION 1.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include("${LINK_DIR}/AbletonLinkConfig.cmake")
add_executable(linkaudio-send src/sender.cpp)
target_link_libraries(linkaudio-send Ableton::Link)
add_executable(linkclock-send src/clock_sender.cpp)
target_link_libraries(linkclock-send Ableton::Link)
add_executable(audiocast-settings src/settings.cpp)
target_link_libraries(audiocast-settings dl)
add_executable(audiocast-session src/session.cpp)
add_executable(audiocast-cksum src/checksum.cpp)
add_executable(alsa-probe src/alsa_probe.cpp)
target_link_libraries(alsa-probe dl)
add_executable(audiocast-core-probe src/core_probe.cpp)
target_link_libraries(audiocast-core-probe dl)
foreach(target audiocast-settings linkclock-send linkaudio-send audiocast-session alsa-probe audiocast-cksum audiocast-core-probe)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_link_options(${target} PRIVATE -static-libstdc++ -static-libgcc)
  endif()
endforeach()
option(AUDIOCAST_CLOCK_TESTS "Build the local Link clock integration-test peer" OFF)
if(AUDIOCAST_CLOCK_TESTS)
  add_executable(link-clock-peer tests/link_peer.cpp)
  target_link_libraries(link-clock-peer Ableton::Link)
  add_executable(link-audio-peer tests/link_audio_peer.cpp)
  target_link_libraries(link-audio-peer Ableton::Link)
endif()
